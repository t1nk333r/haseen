# desktop — the Hyprland session and the few tools its default binds call.
# CachyOS already ships the audio stack; --needed makes those lines no-ops.
# greetd/tuigreet and GPU extras are decided at apply time (layer.sh), not here.
hyprland                     # the compositor (Lua config, 0.55+)
uwsm                         # session manager: systemd units, env.d, graphical-session.target
xdg-desktop-portal-hyprland  # screenshare/screencast portal for Hyprland
xdg-desktop-portal-gtk       # file chooser and settings portals (the hyprland portal has none)
pipewire                     # audio/video server
pipewire-pulse               # PulseAudio clients (most apps) on PipeWire
wireplumber                  # session manager for PipeWire; wpctl drives the volume keys
quickshell                   # runtime for the haseen shell (layers/shell)
foot                         # default terminal: the lightest Wayland terminal that is complete
brightnessctl                # brightness keys
playerctl                    # media keys (MPRIS)
wl-clipboard                 # wl-copy: screenshots to the clipboard
grim                         # screenshots
slurp                        # region selection for screenshots
hyprpicker                   # colour picker bind
polkit                       # privilege prompts; the shell's haseen.polkit service is the agent
gnome-keyring                # Secret Service for browsers/apps, unlocked by PAM at login (keepassxc is a GUI app, not a session service)
xdg-utils                    # xdg-open / xdg-settings: the browser bind opens the default browser
# The default handlers in default/applications/mimeapps.list name these, so
# they ship with the session: a vendor default that points at a missing app is
# worse than none. xdg-terminal-exec is how a Terminal=true entry (yazi, nvim)
# and GTK's "Open in Terminal" reach $TERMINAL.
aur:xdg-terminal-exec        # only in [omarchy]; flea depends on it too
yazi                         # default inode/directory handler (TUI, Terminal=true)
imv                          # default image/* handler (pulls libheif for HEIC/AVIF)
mpv                          # default video/* and application/ogg handler
papers                       # default application/pdf handler
qt6-wayland                  # native Wayland for Qt 6 apps (Quickshell included)
inter-font                   # shell UI font (theme token fontFamily)
ttf-jetbrains-mono-nerd      # terminal and shell mono font with icon glyphs (fontMono)
noto-fonts-emoji             # colour emoji; noto-fonts itself comes from base
gpu-screen-recorder          # haseen capture screenrecord (drives the red privacy dot)
tesseract                    # haseen capture text (OCR)
tesseract-data-eng           # OCR language data
tesseract-data-ara           # Arabic OCR; -l ara+eng is what mixed screen text needs
satty                        # annotation editor for haseen capture screenshot --edit
zbar                         # haseen capture qr
libnotify                    # notify-send for capture/reminder/toggle feedback
hyprsunset                   # night light (haseen.nightlight)
aur:ttfx                     # Omarchy's screensaver text effects, Rust port (owner choice over python tte)
aur:bibata-cursor-theme      # cursor Bibata-Modern-Ice (owner's luna default); prebuilt in Chaotic-AUR
power-profiles-daemon        # haseen powerprofile {list,set}: AC vs battery profile
lsp-plugins-lv2              # lookahead limiter every speaker tuning ends in (haseen audio tuning)
fcitx5                       # input method: Arabic and English on one key
fcitx5-configtool            # fcitx5 GUI, the only sane way to edit a layout list
fcitx5-qt                    # Qt 6 IME module (Quickshell, Qt apps)
fcitx5-gtk                   # GTK IME module
plymouth                     # boot splash and the LUKS password prompt (haseen plymouth set)
